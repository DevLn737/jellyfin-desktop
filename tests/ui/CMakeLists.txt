cmake_minimum_required(VERSION 3.31.6)
project(WindowCloseTests LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../build-manifest.json" build_manifest)
string(JSON locked_qt_version GET "${build_manifest}" qt version)
find_package(Qt6 ${locked_qt_version} EXACT REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(window_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/ui/KonvergoWindow.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${window_source}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract-close-methods.py"
  "${window_source}" "${CMAKE_CURRENT_BINARY_DIR}/WindowCloseMethods.inc"
  RESULT_VARIABLE extraction_result)
if(NOT extraction_result EQUAL 0)
  message(FATAL_ERROR "Could not extract the actual production shutdown methods")
endif()
add_executable(window-close-tests WindowCloseTest.cpp)
target_compile_features(window-close-tests PRIVATE cxx_std_17)
target_include_directories(window-close-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(window-close-tests PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(window-close-tests PRIVATE /utf-8)
endif()
enable_testing()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract-fullscreen-methods.py"
  "${window_source}" "${CMAKE_CURRENT_BINARY_DIR}/WindowFullscreenMethods.inc"
  RESULT_VARIABLE fullscreen_extraction_result)
if(NOT fullscreen_extraction_result EQUAL 0)
  message(FATAL_ERROR "Could not extract the production fullscreen methods")
endif()
add_executable(window-fullscreen-tests WindowFullscreenTest.cpp)
target_compile_features(window-fullscreen-tests PRIVATE cxx_std_17)
target_include_directories(window-fullscreen-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(window-fullscreen-tests PRIVATE Qt6::Core)
foreach(scenario windowed-setting enter-exit enter-exit-enter screen-added-exit always-fullscreen openelec maximized-roundtrips windowed-roundtrips)
  add_test(NAME "window-fullscreen-${scenario}" COMMAND window-fullscreen-tests "${scenario}")
  set_tests_properties("window-fullscreen-${scenario}" PROPERTIES TIMEOUT 5)
endforeach()
foreach(scenario native-close direct-quit reentrant-save duplicate-close fullscreen-close fullscreen-quit)
  add_test(NAME "window-close-${scenario}" COMMAND window-close-tests "${scenario}")
  set_tests_properties("window-close-${scenario}" PROPERTIES TIMEOUT 5)
endforeach()
