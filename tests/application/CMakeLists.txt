cmake_minimum_required(VERSION 3.31.6)
project(ApplicationScopeTests LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../build-manifest.json" build_manifest)
string(JSON locked_qt_version GET "${build_manifest}" qt version)
find_package(Qt6 ${locked_qt_version} EXACT REQUIRED COMPONENTS Core)
add_executable(application-scope-tests ApplicationScopeTest.cpp)
target_compile_features(application-scope-tests PRIVATE cxx_std_17)
target_include_directories(application-scope-tests PRIVATE ../../src)
target_link_libraries(application-scope-tests PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(application-scope-tests PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME application-scope COMMAND application-scope-tests)
