cmake_minimum_required(VERSION 3.31.6)
project(DesktopDiagnosticsTests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Test)
enable_testing()
add_executable(diagnostics-tests DiagnosticsTest.cpp
  ../../src/diagnostics/DiagnosticsBuffer.cpp ../../src/utils/CachedRegexMatcher.cpp)
target_include_directories(diagnostics-tests PRIVATE ../../src)
target_link_libraries(diagnostics-tests PRIVATE Qt6::Core Qt6::Test)
if(MSVC)
  target_compile_options(diagnostics-tests PRIVATE /utf-8)
endif()
add_test(NAME bounded-private-diagnostics COMMAND diagnostics-tests)
