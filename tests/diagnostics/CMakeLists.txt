cmake_minimum_required(VERSION 3.31.6)
project(DesktopDiagnosticsTests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Test Concurrent)
enable_testing()
add_executable(diagnostics-tests DiagnosticsTest.cpp
  ../../src/diagnostics/DiagnosticsBuffer.cpp ../../src/utils/CachedRegexMatcher.cpp)
target_include_directories(diagnostics-tests PRIVATE ../../src)
target_link_libraries(diagnostics-tests PRIVATE Qt6::Core Qt6::Test)
if(MSVC)
  target_compile_options(diagnostics-tests PRIVATE /utf-8)
endif()
add_test(NAME bounded-private-diagnostics COMMAND diagnostics-tests)

add_executable(process-sampler-tests ProcessSamplerTest.cpp
  ../../src/diagnostics/ProcessSampler.cpp ../../src/diagnostics/ProcessSampler.h)
target_include_directories(process-sampler-tests PRIVATE ../../src)
target_link_libraries(process-sampler-tests PRIVATE Qt6::Core Qt6::Test Qt6::Concurrent)
if(WIN32)
  target_link_libraries(process-sampler-tests PRIVATE psapi)
endif()
if(MSVC)
  target_compile_options(process-sampler-tests PRIVATE /utf-8)
endif()
add_test(NAME asynchronous-private-process-sampling COMMAND process-sampler-tests)
