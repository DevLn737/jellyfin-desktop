cmake_minimum_required(VERSION 3.31.6)
project(LogRedactionTests LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../build-manifest.json" build_manifest)
string(JSON locked_qt_version GET "${build_manifest}" qt version)
find_package(Qt6 ${locked_qt_version} EXACT REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(log_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/utils/Log.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${log_source}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract-log-method.py"
  "${log_source}" "${CMAKE_CURRENT_BINARY_DIR}/LogCensorMethod.inc"
  RESULT_VARIABLE extraction_result)
if(NOT extraction_result EQUAL 0)
  message(FATAL_ERROR "Could not extract the production log censor method")
endif()
add_executable(log-redaction-tests LogRedactionTest.cpp)
target_compile_features(log-redaction-tests PRIVATE cxx_std_17)
target_include_directories(log-redaction-tests PRIVATE ../../src "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(log-redaction-tests PRIVATE Qt6::Core)
enable_testing()
add_test(NAME known-credential-log-redaction COMMAND log-redaction-tests)
set_tests_properties(known-credential-log-redaction PROPERTIES TIMEOUT 10)
